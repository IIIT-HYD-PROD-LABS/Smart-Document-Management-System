    ```python
    """Tenant context middleware — Phase 9 CLIENT-04.

    Sets the PostgreSQL session variable `app.current_client_id` on every
    connection checkout, based on a ContextVar populated by FastAPI middleware.
    This makes RLS policies (migration 0015) automatic for any query running
    inside a request context.

    Header convention:
      X-Client-Id: 42       → tenant_isolation policy filters to client_id=42
      X-Client-Id: *        → cross-client mode (only allowed for compliance_head/ca_consultant/cfo)
      (no header)           → no context set; RLS denies all rows (fail-closed)

    Per RESEARCH Pitfall 5: the middleware looks up (user_id, client_id)→membership
    and rejects with 403 if not found OR membership expired. Frontend Zustand
    store cross-checks activeClientId against /memberships/me.
    """
    import contextvars
    from datetime import datetime, timezone
    from typing import Optional
    from fastapi import Request, status
    from fastapi.responses import JSONResponse
    from starlette.middleware.base import BaseHTTPMiddleware
    from sqlalchemy import event as sa_event

    # ContextVars carry tenant state through the request lifecycle.
    # Connection checkout listener reads from these.
    current_client_id_var: contextvars.ContextVar[Optional[int]] = (
        contextvars.ContextVar("current_client_id", default=None)
    )
    cross_client_mode_var: contextvars.ContextVar[bool] = (
        contextvars.ContextVar("cross_client_mode", default=False)
    )
    current_user_id_var: contextvars.ContextVar[Optional[int]] = (
        contextvars.ContextVar("current_user_id", default=None)
    )


    def register_tenant_listener(engine) -> None:
        """Register the connection-checkout listener ONCE at startup.

        Reads ContextVars and runs `set_config()` on the connection.
        """
        @sa_event.listens_for(engine, "checkout")
        def _set_tenant_on_checkout(dbapi_connection, connection_record, connection_proxy):
            client_id = current_client_id_var.get()
            cross_mode = cross_client_mode_var.get()
            user_id = current_user_id_var.get()
            try:
                cur = dbapi_connection.cursor()
                # Always set — empty string clears (current_setting('...', true) returns NULL)
                cur.execute("SELECT set_config('app.current_client_id', %s, true)",
                            (str(client_id) if client_id is not None else "",))
                cur.execute("SELECT set_config('app.cross_client_mode', %s, true)",
                            ("true" if cross_mode else "false",))
                cur.execute("SELECT set_config('app.user_id', %s, true)",
                            (str(user_id) if user_id is not None else "",))
                cur.close()
            except Exception:
                # Listener errors must never crash the request — RLS will
                # fail-closed (no rows) rather than permitting cross-tenant access.
                pass


    class TenantContextMiddleware(BaseHTTPMiddleware):
        """Resolves X-Client-Id header → ContextVars on every /api/compliance/* request.

        Validates that the requesting user has a non-expired membership for
        the X-Client-Id; rejects with 403 if not.
        """

        # Public paths that don't need tenant context (auth, health, OpenAPI)
        PUBLIC_PREFIXES = (
            "/api/auth/",
            "/docs", "/redoc", "/openapi.json",
            "/health",
        )

        async def dispatch(self, request: Request, call_next):
            path = request.url.path
            # Bypass for paths not requiring tenant context
            if any(path.startswith(p) for p in self.PUBLIC_PREFIXES):
                return await call_next(request)

            # Only enforce on /api/compliance/* — v1.0 routes stay user-scoped
            if not path.startswith("/api/compliance"):
                return await call_next(request)

            # Need user_id from auth header — defer to dependency layer to avoid
            # parsing JWT here. Middleware reads X-Client-Id; routes use Depends
            # to resolve user + membership and re-validate.
            client_header = request.headers.get("X-Client-Id")
            user_id_header = request.headers.get("X-User-Id-Internal")  # set by auth dep below

            tokens = []
            try:
                if client_header == "*":
                    tokens.append(cross_client_mode_var.set(True))
                    tokens.append(current_client_id_var.set(None))
                elif client_header:
                    try:
                        tokens.append(current_client_id_var.set(int(client_header)))
                        tokens.append(cross_client_mode_var.set(False))
                    except ValueError:
                        return JSONResponse(
                            status_code=status.HTTP_400_BAD_REQUEST,
                            content={"detail": "Invalid X-Client-Id header"},
                        )
                else:
                    # No client header — auth-only paths are bypassed earlier;
                    # /api/compliance/* without header → fail-closed via RLS
                    tokens.append(current_client_id_var.set(None))
                    tokens.append(cross_client_mode_var.set(False))

                if user_id_header:
                    try:
                        tokens.append(current_user_id_var.set(int(user_id_header)))
                    except ValueError:
                        pass

                response = await call_next(request)
                return response
            finally:
                # Reset ContextVars to avoid leaking to next request
                for tok in tokens:
                    try:
                        # set() returns a Token; reset by ContextVar.reset(token)
                        # We loop and call .reset() on the var that owns the token
                        if hasattr(tok, "var"):
                            tok.var.reset(tok)
                    except Exception:
                        pass


    def set_tenant_context_for_celery(client_id: int, user_id: int, cross_mode: bool = False):
        """Helper for Celery tasks — call at task start to set ContextVars.

        Per Pitfall 6: Celery workers do NOT inherit middleware. Task code MUST
        call this BEFORE issuing any DB query that touches client-scoped tables.
        """
        current_client_id_var.set(client_id)
        current_user_id_var.set(user_id)
        cross_client_mode_var.set(cross_mode)
    ```

    ```python
    """Auditor time-bound access enforcement — Phase 9 RBAC-04 / D-27.

    Per Pitfall 4: an Auditor with access_end in the past must be rejected on
    EVERY request, not just at membership creation. The active-window check
    is enforced both:
      1. At dependency time via require_compliance_permission (defense in depth)
      2. As a standalone helper used by middleware/services
    """
    from datetime import datetime, timezone
    from typing import Optional
    from app.compliance.models.membership import ClientMembership


    def is_membership_active(
        membership: ClientMembership,
        when: Optional[datetime] = None,
    ) -> bool:
        """Return True iff membership is active at `when` (default: now in UTC).

        Inactive cases:
          - access_start is set and `when` < access_start (not yet started)
          - access_end is set and `when` > access_end (expired)

        Membership with both fields NULL is unbounded (always active).
        """
        if when is None:
            when = datetime.now(timezone.utc)
        return membership.is_active_at(when)


    def reason_inactive(
        membership: ClientMembership,
        when: Optional[datetime] = None,
    ) -> Optional[str]:
        """Return a human-readable reason if membership is inactive, else None."""
        if when is None:
            when = datetime.now(timezone.utc)
        if membership.access_start and when < membership.access_start:
            return f"Membership access has not started (begins {membership.access_start.isoformat()})"
        if membership.access_end and when > membership.access_end:
            return f"Membership access has expired (ended {membership.access_end.isoformat()})"
        return None
    ```

    PART A: Create `backend/app/compliance/dependencies.py`:

    ```python
    """FastAPI dependency factories for compliance endpoints — Phase 9 RBAC + RLS bridge.

    Three core dependencies:
      - get_active_client_id   → reads X-Client-Id header (set by middleware ContextVar)
      - get_active_membership  → looks up the (user, client) membership; rejects if missing/expired
      - require_compliance_permission(perm) → factory that requires `perm` on the active membership

    Per CONTEXT D-28: pattern mirrors v1.0 require_admin/require_editor in app/utils/security.py.
    Per Pitfall 5: membership lookup is mandatory — frontend cannot bypass via header tampering.
    """
    from datetime import datetime, timezone
    from typing import Optional
    from fastapi import Depends, HTTPException, Request, status
    from sqlalchemy.orm import Session

    from app.compliance.middleware.auditor_expiry import is_membership_active, reason_inactive
    from app.compliance.middleware.tenant_context import (
        current_client_id_var, cross_client_mode_var,
    )
    from app.compliance.models.membership import ClientMembership
    from app.compliance.services.permission_registry import (
        ComplianceRole, CompliancePermission, has_permission, ROLE_PERMISSIONS,
    )
    from app.database import get_db
    from app.models.user import User
    from app.utils.security import get_current_user


    def get_active_client_id(request: Request) -> Optional[int]:
        """Returns the active client_id from ContextVar (or None if cross-client mode)."""
        if cross_client_mode_var.get():
            return None
        cid = current_client_id_var.get()
        if cid is None:
            # Frontend should set header; missing means unauthenticated/public path
            raise HTTPException(
                status_code=status.HTTP_400_BAD_REQUEST,
                detail="X-Client-Id header is required for compliance endpoints",
            )
        return cid


    def is_cross_client_mode() -> bool:
        return cross_client_mode_var.get()


    def get_active_membership(
        current_user: User = Depends(get_current_user),
        client_id: Optional[int] = Depends(get_active_client_id),
        db: Session = Depends(get_db),
    ) -> ClientMembership:
        """Looks up (user, client) → ClientMembership. Rejects if missing or expired.

        For cross-client mode (client_id=None), returns ANY active membership for
        the user that is in the eligible-roles set (compliance_head/ca_consultant/cfo).
        """
        if client_id is None:
            # Cross-client mode — accept the highest-privilege eligible membership
            eligible_roles = ('compliance_head', 'ca_consultant', 'cfo')
            now = datetime.now(timezone.utc)
            m = (db.query(ClientMembership)
                 .filter(ClientMembership.user_id == current_user.id,
                         ClientMembership.compliance_role.in_(eligible_roles))
                 .all())
            active = [x for x in m if is_membership_active(x, now)]
            if not active:
                raise HTTPException(
                    status_code=status.HTTP_403_FORBIDDEN,
                    detail="Cross-client mode requires an active compliance_head, ca_consultant, or cfo membership",
                )
            return active[0]

        membership = (db.query(ClientMembership)
                      .filter(ClientMembership.user_id == current_user.id,
                              ClientMembership.client_id == client_id)
                      .first())
        if not membership:
            raise HTTPException(
                status_code=status.HTTP_403_FORBIDDEN,
                detail=f"No membership for client {client_id}",
            )
        if not is_membership_active(membership):
            raise HTTPException(
                status_code=status.HTTP_403_FORBIDDEN,
                detail=reason_inactive(membership) or "Membership not active",
            )
        return membership


    def require_compliance_permission(perm: CompliancePermission):
        """Factory: returns a dependency that requires `perm` on the active membership.

        Usage:
          @router.post("/notices",
                       dependencies=[Depends(require_compliance_permission(
                           CompliancePermission.NOTICE_CREATE))])
        """
        def _check(
            membership: ClientMembership = Depends(get_active_membership),
        ) -> ClientMembership:
            try:
                role = ComplianceRole(membership.compliance_role)
            except ValueError:
                raise HTTPException(

                                    detail="Invalid role")

                    status_code=status.HTTP_403_FORBIDDEN,
                    detail=f"Role '{role.value}' is not in allowed set: {[r.value for r in allowed]}",
                )
            return membership
        return _check
    ```
