    from fastapi import APIRouter, Depends, HTTPException, status
    from sqlalchemy.orm import Session

    from app.compliance.dependencies import require_compliance_permission
    from app.compliance.models.membership import ClientMembership
    from app.compliance.schemas.client import MembershipCreate, MembershipOut
    from app.compliance.services.permission_registry import CompliancePermission
    from app.database import get_db
    from app.models.user import User
    from app.utils.security import get_current_user
    from app.services.audit_service import log_audit_event


    router = APIRouter(prefix="/clients/{client_id}/memberships",
                       tags=["compliance-memberships"])


    @router.post("", response_model=MembershipOut, status_code=status.HTTP_201_CREATED)
    def add_member(
        client_id: int,
        payload: MembershipCreate,
        current_user: User = Depends(get_current_user),
        db: Session = Depends(get_db),
        _gate = Depends(require_compliance_permission(
            CompliancePermission.CLIENT_MANAGE_TEAM,
        )),
    ):
        existing = (db.query(ClientMembership)
                    .filter(ClientMembership.user_id == payload.user_id,
                            ClientMembership.client_id == client_id).first())
        if existing:
            raise HTTPException(409, "User already has a membership for this client")
        m = ClientMembership(
            user_id=payload.user_id,
            client_id=client_id,
            compliance_role=payload.compliance_role,
            access_start=payload.access_start,
            access_end=payload.access_end,
        )
        db.add(m)
        db.commit()
        db.refresh(m)
        log_audit_event(
            user_id=current_user.id,
            action="membership_added",
            resource_type="ClientMembership",
            resource_id=m.id,
            details={
                "client_id": client_id, "user_id": payload.user_id,
                "compliance_role": payload.compliance_role,
            },
        )
        return m


    @router.delete("/{membership_id}", status_code=status.HTTP_204_NO_CONTENT)
    def remove_member(
        client_id: int,
        membership_id: int,
        current_user: User = Depends(get_current_user),
        db: Session = Depends(get_db),
        _gate = Depends(require_compliance_permission(
            CompliancePermission.CLIENT_MANAGE_TEAM,
        )),
    ):
        m = (db.query(ClientMembership)
             .filter(ClientMembership.id == membership_id,
                     ClientMembership.client_id == client_id).first())
        if not m:
            raise HTTPException(404, "Membership not found")
        revoked_user = m.user_id
        revoked_role = m.compliance_role
        db.delete(m)
        db.commit()
        log_audit_event(
            user_id=current_user.id,
            action="membership_removed",
            resource_type="ClientMembership",
            resource_id=membership_id,
            details={
                "client_id": client_id, "user_id": revoked_user,
                "compliance_role": revoked_role,
            },
        )

    def _permission_for_target_status(target: NoticeStatus) -> CompliancePermission:
        """Map target status → required permission."""
        if target == NoticeStatus.SUBMITTED:
            return CompliancePermission.NOTICE_SUBMIT
        if target in (NoticeStatus.RESOLVED, NoticeStatus.DISMISSED):
            return CompliancePermission.NOTICE_APPROVE
        # Forward transitions to under_review / response_drafted
        return CompliancePermission.NOTICE_DRAFT_RESPONSE


    @router.patch("/{notice_id}/status", response_model=NoticeOut)
    def transition_status(
        notice_id: int,
        payload: NoticeStatusTransition,
        current_user: User = Depends(get_current_user),
        membership = Depends(get_active_membership),
        db: Session = Depends(get_db),
    ):
        """LIFE-04: transition notice through state machine.

        Permission depends on target status (see _permission_for_target_status).
        Returns 422 with valid_next_statuses on InvalidTransitionError.
        """
        target = NoticeStatus(payload.new_status)
        # Manual permission gate based on target
        from app.compliance.services.permission_registry import has_permission, ComplianceRole
        try:
            role = ComplianceRole(membership.compliance_role)
        except ValueError:
            raise HTTPException(403, "Invalid role")
        required = _permission_for_target_status(target)
        if not has_permission(role, required):
            raise HTTPException(403, f"Role '{role.value}' lacks '{required.value}' for transition to '{target.value}'")
        try:
            n = transition_notice_status(db, notice_id, target, current_user, payload.reason)
        except InvalidTransitionError as e:
            valid = [s.value for s in ALLOWED_TRANSITIONS.get(NoticeStatus(
                db.query(ComplianceNotice).filter(ComplianceNotice.id == notice_id).one().status), [])]
            raise HTTPException(
                status_code=status.HTTP_422_UNPROCESSABLE_ENTITY,
                detail={"message": str(e), "valid_next_statuses": valid},
            )
        return n


    @router.post("/bulk", response_model=BulkUpdateResponse)
    def bulk_update(
        payload: BulkUpdateRequest,
        current_user: User = Depends(get_current_user),
        db: Session = Depends(get_db),
        _gate = Depends(require_compliance_permission(CompliancePermission.NOTICE_BULK_UPDATE)),
    ):
        """LIFE-08: bulk status update with partial-failure semantics."""
        target = NoticeStatus(payload.new_status)
        result = bulk_update_status(
            db=db, notice_ids=payload.notice_ids,
            new_status=target, user=current_user, reason=payload.reason,
        )
        return result

    def upload_notice_file(
        notice_id: int,
        file: UploadFile = File(...),
        current_user: User = Depends(get_current_user),
        db: Session = Depends(get_db),
        membership = Depends(require_compliance_permission(CompliancePermission.NOTICE_CREATE)),
    ):
        """LIFE-01, LIFE-02: upload a notice PDF/JPG/PNG.

        Reuses v1.0 storage_service + Document model (D-10). The Document row
        gets notice_id FK; OCR pipeline runs as before via Celery (existing v1.0 path).
        """
        n = db.query(ComplianceNotice).filter(ComplianceNotice.id == notice_id).first()
        if not n:
            raise HTTPException(404, "Notice not found")
        # Validate file type
        if file.content_type not in ("application/pdf", "image/jpeg", "image/png", "image/jpg"):
            raise HTTPException(400, "Only PDF, JPG, PNG accepted")
        contents = file.file.read()
        # Save via existing v1.0 helper (returns dict with file_path/s3_url)
        saved = save_uploaded_file(file_bytes=contents, original_filename=file.filename or "notice")
        d = Document(
            user_id=current_user.id,
            filename=saved.get("filename", file.filename or "notice"),
            original_filename=file.filename or "notice",
            file_type=(file.content_type or "").split("/")[-1] or "pdf",
            file_size=len(contents),
            file_path=saved.get("file_path"),
            s3_url=saved.get("s3_url"),
            status=DocumentStatus.PENDING,
            notice_id=notice_id,
        )
        db.add(d)
        db.commit()
        db.refresh(d)
        # Link the document_id back on the notice (the FIRST upload becomes the notice's primary file)
        if n.document_id is None:
            n.document_id = d.id
            db.commit()
            db.refresh(n)
        # Log to user-facing activity
        log_activity(
            db, notice_id=notice_id, user_id=current_user.id, type="file_attached",
            details={"document_id": d.id, "filename": d.original_filename, "size": d.file_size},
        )
        db.commit()
        return n
