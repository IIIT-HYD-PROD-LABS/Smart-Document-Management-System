    ```typescript
    "use client";
    import { FiInbox, FiEye, FiEdit3, FiSend, FiCheckCircle, FiXCircle, FiAlertTriangle } from "react-icons/fi";
    import type { NoticeStatus } from "@/types/compliance";

    const STATUS_CONFIG: Record<NoticeStatus, { color: string; icon: React.ComponentType<{ className?: string }>; label: string }> = {
      received: { color: "#3b82f6", icon: FiInbox, label: "Received" },
      under_review: { color: "#f59e0b", icon: FiEye, label: "Under Review" },
      response_drafted: { color: "#8b5cf6", icon: FiEdit3, label: "Response Drafted" },
      submitted: { color: "#06b6d4", icon: FiSend, label: "Submitted" },
      resolved: { color: "#10b981", icon: FiCheckCircle, label: "Resolved" },
      dismissed: { color: "#ef4444", icon: FiXCircle, label: "Dismissed" },
    };

    export function StatusPill({ status, overdue = false, size = "sm" }: { status: NoticeStatus; overdue?: boolean; size?: "sm" | "md" }) {
      const c = STATUS_CONFIG[status];
      const Icon = c.icon;
      const padding = size === "md" ? "px-3 py-1.5" : "px-2 py-1";
      const text = size === "md" ? "text-[13px]" : "text-[11px]";
      return (
        <span className="inline-flex items-center gap-1.5">
          {overdue && <FiAlertTriangle className="w-3.5 h-3.5 text-[#ef4444]" aria-label="Overdue" />}
          <span className={`inline-flex items-center gap-1 ${padding} ${text} rounded font-medium`}
            style={{ backgroundColor: `${c.color}1a`, color: c.color }} aria-label={`Status: ${c.label}`}>
            <Icon className="w-3 h-3" />
            {c.label}
          </span>
        </span>
      );
    }

    export { STATUS_CONFIG };
    ```
