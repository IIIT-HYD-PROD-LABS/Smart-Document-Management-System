    ```python
    """Phase 15 v2.0 end-to-end smoke — Gmail MCP + Bills.

    Mocks Gmail API; uses real DB + Redis. Mirrors smoke_phase{10,12,13}_v20.py.

    Usage:
        cd backend && python ../scripts/smoke_phase15_v20.py
    """
    from __future__ import annotations

    import asyncio
    import os
    import sys
    from datetime import date, timedelta
    from decimal import Decimal
    from unittest.mock import patch

    sys.path.insert(0, os.path.join(os.path.dirname(__file__), "..", "backend"))

    GREEN = "\033[92m"
    RED = "\033[91m"
    YELLOW = "\033[93m"
    RESET = "\033[0m"


    def passed(name: str) -> None:
        print(f"{GREEN}[PASS]{RESET} {name}")


    def failed(name: str, reason: str) -> None:
        print(f"{RED}[FAIL]{RESET} {name}: {reason}")
        sys.exit(1)


    def info(msg: str) -> None:
        print(f"{YELLOW}[INFO]{RESET} {msg}")


    def main() -> None:
        info("Phase 15 v2.0 smoke starting...")

        # --- 1. alembic head ---
        try:
            from alembic.config import Config
            from alembic.script import ScriptDirectory
            from sqlalchemy import create_engine, text
            cfg = Config(os.path.join(os.path.dirname(__file__), "..", "backend", "alembic.ini"))
            scripts = ScriptDirectory.from_config(cfg)
            head = scripts.get_current_head()
            db_url = os.environ.get("DATABASE_URL_RUNTIME") or os.environ["DATABASE_URL"]
            with create_engine(db_url).connect() as conn:
                actual = conn.execute(text("SELECT version_num FROM alembic_version LIMIT 1")).scalar()
            assert actual == head == "0025", f"head={head} actual={actual}"
            passed(f"alembic_head (0025)")
        except Exception as e:
            failed("alembic_head", str(e))

        # --- 2. tables exist ---
        try:
            from sqlalchemy import inspect
            insp = inspect(create_engine(db_url))
            tables = set(insp.get_table_names())
            expected = {"gmail_credentials", "gmail_filter_rules", "gmail_message_log", "gmail_fetch_log", "bills"}
            missing = expected - tables
            assert not missing, f"missing: {missing}"
            passed("tables_exist (5 phase 15 tables)")
        except Exception as e:
            failed("tables_exist", str(e))

        # --- 3. documents.source_email_id ---
        try:
            cols = {c["name"] for c in insp.get_columns("documents")}
            assert "source_email_id" in cols
            passed("document_source_email_id_column")
        except Exception as e:
            failed("document_source_email_id_column", str(e))

        # --- 4. filter_rules.priority ---
        try:
            cols = {c["name"] for c in insp.get_columns("gmail_filter_rules")}
            assert "priority" in cols
            passed("filter_rule_priority_column (open question #5)")
        except Exception as e:
            failed("filter_rule_priority_column", str(e))

        # --- 5. partial unique index ---
        try:
            idxs = insp.get_indexes("bills")
            partial = [i for i in idxs if i["name"] == "ux_bills_recurrence_key"]
            assert partial, "ux_bills_recurrence_key not found"
            passed("partial_unique_index (Pitfall 8)")
        except Exception as e:
            failed("partial_unique_index", str(e))

        # --- 6. classifier 4 cases ---
        try:
            from app.email.services.classifier import classify
            assert classify("user@cbic-gst.gov.in", "Show Cause Notice u/s 73") == (True, 1.0), "gov.in + keyword"
            assert classify("regulatory@rbi.org.in", "Penalty Hearing") == (True, 1.0), "rbi.org.in (reconciliation #4)"
            assert classify("user@cbic-gst.gov.in", "Quarterly Newsletter") == (False, 0.5), "review queue"
            assert classify("advocate@gmail.com", "Show Cause Forwarded") == (False, 0.0), "D-33 forwarded -> dms_only"
            passed("classifier_4_cases (incl. reconciliation #4)")
        except Exception as e:
            failed("classifier_4_cases", str(e))

        # --- 7. mcp 6 tools (in-memory) ---
        try:
            from fastmcp import Client
            from app.email.mcp.server import mcp

            async def _list_tools():
                async with Client(mcp) as c:
                    return await c.list_tools()
            tools = asyncio.run(_list_tools())
            tool_names = {t.name for t in tools}
            expected = {"gmail_search", "gmail_read_message", "gmail_list_attachments", "gmail_get_attachment", "gmail_list_labels", "gmail_modify_labels"}
            missing = expected - tool_names
            assert not missing, f"missing tools: {missing}"
            passed("mcp_six_tools (in-memory transport, reconciliation #1)")
        except Exception as e:
            failed("mcp_six_tools", str(e))

        # --- 8. fernet round-trip ---
        try:
            from app.compliance.utils.pii_encryption import decrypt_field, encrypt_field
            plain = "test-refresh-token-xyz"
            enc = encrypt_field(plain)
            dec = decrypt_field(enc).decode("utf-8")
            assert dec == plain, f"got {dec}"
            passed("fernet_round_trip")
        except Exception as e:
            failed("fernet_round_trip", str(e))

        # --- 9-12 require a DB session + fixture rows; mock alert dispatch ---
        try:
            from app.compliance.middleware.tenant_context import set_tenant_context_for_celery
            from app.database import SessionLocal
            from app.email.models.bill import Bill
            from app.email.models.credential import GmailCredential
            from app.email.models.filter_rule import GmailFilterRule
            from app.email.services.bill_service import mark_paid

            db = SessionLocal()
            set_tenant_context_for_celery(client_id=None, user_id=None, cross_mode=True)
            existing_client = db.execute(text("SELECT id FROM compliance_clients LIMIT 1")).scalar()
            existing_user = db.execute(text("SELECT id FROM users LIMIT 1")).scalar()
            if existing_client is None or existing_user is None:
                info("No compliance_clients or users seeded; skipping checks 9-12 (run after Phase 9 client onboarding fixtures)")
                passed("smoke_partial_completion")
                return
            set_tenant_context_for_celery(client_id=existing_client, user_id=existing_user, cross_mode=False)

            # --- 9. bill mark_paid + audit ---
            bill = Bill(
                client_id=existing_client,
                user_id=existing_user,
                biller_name="Smoke Test Biller",
                biller_name_normalized="smoke test biller",
                biller_category="utility",
                amount_due=Decimal("100.00"),
                due_date=date.today() + timedelta(days=5),
                payment_status="pending",
            )
            db.add(bill)
            db.commit()
            db.refresh(bill)
            with patch("app.services.audit_service.log_audit_event_strict") as mock_audit:
                paid_bill = mark_paid(
                    db, bill_id=bill.id, payment_date=date.today(),
                    payment_reference="UPI-TEST-123", payment_method="upi",
                    user_id=existing_user,
                )
                assert paid_bill.payment_status == "paid"
                assert mock_audit.called, "audit log not invoked"
                # W5: robust to either keyword OR positional 'action' args.
                call_args = mock_audit.call_args
                action_in_kwargs = call_args.kwargs.get("action") == "BILL_MARK_PAID"
                action_in_repr = "BILL_MARK_PAID" in str(call_args)
                assert action_in_kwargs or action_in_repr, (
                    f"BILL_MARK_PAID not found in mark_paid audit call: {call_args}"
                )
            db.delete(bill)
            db.commit()
            passed("bill_mark_paid_audit")

            # --- 10. filter rule priority ordering ---
            cred = GmailCredential(
                user_id=existing_user, client_id=existing_client,
                refresh_token_enc=b"\x00" * 16, status="active",
            )
            db.add(cred)
            db.commit()
            db.refresh(cred)
            db.add_all([
                GmailFilterRule(credential_id=cred.id, priority=50, route_to="bill"),
                GmailFilterRule(credential_id=cred.id, priority=100, route_to="ignore"),
            ])
            db.commit()
            rules = (
                db.query(GmailFilterRule)
                .filter(GmailFilterRule.credential_id == cred.id)
                .order_by(GmailFilterRule.priority.asc())
                .all()
            )
            assert rules[0].priority == 50, f"first priority {rules[0].priority}"
            db.query(GmailFilterRule).filter(GmailFilterRule.credential_id == cred.id).delete()
            db.delete(cred)
            db.commit()
            passed("filter_rule_priority_ordering")

            # --- 11. bill recurrence with NULL last4 (Pitfall 8) ---
            b1 = Bill(client_id=existing_client, user_id=existing_user, biller_name="Tata Power", biller_name_normalized="tata power", biller_category="utility", amount_due=Decimal("500.00"), payment_status="pending", account_number_last4=None)
            b2 = Bill(client_id=existing_client, user_id=existing_user, biller_name="Tata Power", biller_name_normalized="tata power", biller_category="utility", amount_due=Decimal("600.00"), payment_status="pending", account_number_last4=None)
            db.add_all([b1, b2])
            db.commit()  # should NOT raise IntegrityError thanks to partial unique WHERE last4 IS NOT NULL
            db.delete(b1)
            db.delete(b2)
            db.commit()
            passed("bill_recurrence_partial_unique (Pitfall 8)")

            # --- 12. /api/email/credentials returns 200 (smoke test in-process) ---
            try:
                from fastapi.testclient import TestClient
                from app.main import app
                # Note: TestClient does not authenticate; expect 401 OR 200 depending on stack.
                # Smoke acceptance is reachable + non-500.
                client = TestClient(app)
                resp = client.get("/api/email/credentials")
                assert resp.status_code in (200, 401, 403), f"unexpected {resp.status_code}: {resp.text[:200]}"
                passed(f"api_credentials_endpoint (status={resp.status_code})")
            except Exception as e:
                failed("api_credentials_endpoint", str(e))

        except Exception as e:
            failed("smoke_steps_9_12", str(e))
        finally:
            try:
                db.close()
            except Exception:
                pass

        info(f"{GREEN}All 12 Phase 15 smoke checks PASSED{RESET}")


    if __name__ == "__main__":
        main()
    ```
