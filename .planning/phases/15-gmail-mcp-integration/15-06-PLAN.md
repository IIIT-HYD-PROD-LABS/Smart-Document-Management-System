    "use client";

    import { useState } from "react";
    import toast from "react-hot-toast";
    import { emailApi, GmailCredentialResponse } from "@/lib/email-api";

    interface Props {
      credential: GmailCredentialResponse | null;
      onChange?: () => void;
    }

    export default function ConnectGmailButton({ credential, onChange }: Props) {
      const [busy, setBusy] = useState(false);

      const handleConnect = async () => {
        setBusy(true);
        try {
          const resp = await emailApi.connectGmail();
          window.location.href = resp.data.authorize_url;
        } catch (e: any) {
          toast.error(e?.response?.data?.detail || "Failed to start OAuth");
          setBusy(false);
        }
      };

      const handleDisconnect = async () => {
        if (!credential) return;
        if (!confirm("Disconnect Gmail? Scanner will stop.")) return;
        setBusy(true);
        try {
          await emailApi.deleteCredential(credential.id);
          toast.success("Gmail disconnected");
          onChange?.();
        } catch (e: any) {
          toast.error(e?.response?.data?.detail || "Disconnect failed");
        } finally {
          setBusy(false);
        }
      };

      if (!credential) {
        return (
          <button
            onClick={handleConnect}
            disabled={busy}
            className="px-4 py-2 bg-primary-500 text-white rounded-lg hover:bg-primary-600 disabled:opacity-50"
          >
            {busy ? "Redirecting..." : "Connect Gmail"}
          </button>
        );
      }

      if (credential.status === "revoked") {
        return (
          <div className="space-y-2">
            <div className="p-3 bg-red-900/20 border border-red-500/40 text-red-200 rounded-lg">
              Gmail credential revoked. Please reconnect to resume scanning.
            </div>
            <button
              onClick={handleConnect}
              disabled={busy}
              className="px-4 py-2 bg-amber-500 text-white rounded-lg hover:bg-amber-600 disabled:opacity-50"
            >
              Reconnect
            </button>
          </div>
        );
      }

      return (
        <div className="flex items-center gap-3">
          <div className="text-emerald-400">
            Connected{credential.google_account_email ? ` to ${credential.google_account_email}` : ""}
          </div>
          <button
            onClick={handleConnect}
            className="px-3 py-1.5 bg-zinc-700 text-white rounded hover:bg-zinc-600 text-sm"
          >
            Reconnect
          </button>
          <button
            onClick={handleDisconnect}
            disabled={busy}
            className="px-3 py-1.5 bg-red-700 text-white rounded hover:bg-red-600 text-sm disabled:opacity-50"
          >
            Disconnect
          </button>
        </div>
      );
    }

    "use client";

    import { useEffect, useState } from "react";
    import toast from "react-hot-toast";
    import { emailApi, GmailFilterRule } from "@/lib/email-api";

    interface Props { credentialId: number }

    export default function FilterRulesEditor({ credentialId }: Props) {
      const [rules, setRules] = useState<GmailFilterRule[]>([]);
      const [loading, setLoading] = useState(true);

      const load = async () => {
        setLoading(true);
        try {
          const resp = await emailApi.listFilterRules(credentialId);
          setRules(resp.data);
        } catch (e: any) {
          toast.error(e?.response?.data?.detail || "Failed to load rules");
        } finally {
          setLoading(false);
        }
      };

      useEffect(() => { load(); }, [credentialId]);

      const handleAdd = async () => {
        try {
          await emailApi.createFilterRule(credentialId, {
            priority: 100, sender_pattern: "", subject_pattern: "",
            route_to: "ignore", enabled: true,
          });
          await load();
        } catch (e: any) {
          toast.error(e?.response?.data?.detail || "Failed");
        }
      };

      const handleDelete = async (id: number) => {
        if (!confirm("Delete this rule?")) return;
        try { await emailApi.deleteFilterRule(id); await load(); }
        catch (e: any) { toast.error(e?.response?.data?.detail || "Failed"); }
      };

      const handleSave = async (rule: GmailFilterRule) => {
        try {
          await emailApi.updateFilterRule(rule.id, {
            priority: rule.priority,
            sender_pattern: rule.sender_pattern,
            subject_pattern: rule.subject_pattern,
            route_to: rule.route_to,
            enabled: rule.enabled,
          });
          toast.success("Saved");
          await load();
        } catch (e: any) { toast.error(e?.response?.data?.detail || "Failed"); }
      };

      if (loading) return <div className="text-zinc-400">Loading rules...</div>;

      return (
        <div className="space-y-4">
          <button onClick={handleAdd} className="px-4 py-2 bg-primary-500 text-white rounded">+ Add Rule</button>
          <table className="w-full text-sm">
            <thead className="text-left text-zinc-400">
              <tr><th>Priority</th><th>Sender</th><th>Subject</th><th>Route</th><th>Enabled</th><th></th></tr>
            </thead>
            <tbody>
              {rules.map(r => (
                <tr key={r.id} className="border-t border-zinc-800">
                  <td><input className="w-16 bg-zinc-900 border border-zinc-700 rounded px-2 py-1" type="number" defaultValue={r.priority} onBlur={e => handleSave({ ...r, priority: parseInt(e.target.value) })} /></td>
                  <td><input className="w-full bg-zinc-900 border border-zinc-700 rounded px-2 py-1" defaultValue={r.sender_pattern || ""} onBlur={e => handleSave({ ...r, sender_pattern: e.target.value || null })} /></td>
                  <td><input className="w-full bg-zinc-900 border border-zinc-700 rounded px-2 py-1" defaultValue={r.subject_pattern || ""} onBlur={e => handleSave({ ...r, subject_pattern: e.target.value || null })} /></td>
                  <td>
                    <select className="bg-zinc-900 border border-zinc-700 rounded px-2 py-1" defaultValue={r.route_to} onChange={e => handleSave({ ...r, route_to: e.target.value as any })}>
                      <option value="compliance_notice">Compliance</option>
                      <option value="bill">Bill</option>
                      <option value="dms_only">DMS only</option>
                      <option value="ignore">Ignore</option>
                    </select>
                  </td>
                  <td><input type="checkbox" defaultChecked={r.enabled} onChange={e => handleSave({ ...r, enabled: e.target.checked })} /></td>
                  <td><button onClick={() => handleDelete(r.id)} className="text-red-400 hover:text-red-300">Delete</button></td>
                </tr>
              ))}
            </tbody>
          </table>
        </div>
      );
    }
